# while the Coinbase bot's primary environment currently pins pandas==3.0.1.

# river==0.23.0's own PyPI metadata only declares numpy/pandas/scipy, but
# river.base.base imports typing_extensions directly. On a clean venv that
# resolves only river's declared dependencies (e.g. `pip install --find-links`
# against a pre-downloaded wheel, or any resolver that doesn't pull every
# transitive extra) this import fails *silently* from the bot's point of
# view: bot/river_online_parameter_learner.py's `_import_river()` catches the
# resulting ImportError and reports `river_native_sidecar_unavailable` with
# no further detail. Verified on Python 3.12 / river 0.23.0 on 2026-06-21.
typing_extensions

# via python-dotenv before reading any governor env settings (ACK, mode,

# `tools/prepare_growbot_river_governor_candidate.py --apply` run through
# this interpreter would silently see governor_enabled=false / mode=report_only
# regardless of what .env actually says, because load_dotenv() would be

python-dotenv==1.2.2
